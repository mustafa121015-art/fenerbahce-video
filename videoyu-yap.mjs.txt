import fs from 'fs';
import path from 'path';
import { execSync } from 'child_process';

const publicDir = path.resolve('public');
const srcDir = path.resolve('src');
const outDir = path.resolve('out');

[publicDir, srcDir, outDir].forEach((dir) => {
  if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
});

console.log('1/3 -> Seslendirme ve Altyapı Hazırlanıyor...');

try {
  await import('edge-tts');
} catch (e) {
  console.log('Ses motoru kuruluyor...');
  execSync('npm install edge-tts', { stdio: 'ignore' });
}

const { EdgeTTS } = await import('edge-tts');

const sesMetni = "1907'de Kadıköy sahilinde doğan bu asil meşale, Harington Kupası ile bağımsızlık ateşine dönüştü. Lefter'den Can Bartu'ya, Kadıköy Mabedi'nde yazılan şanlı tarih... Dünden yarına, sonsuza dek Fenerbahçe!";
const tts = new EdgeTTS({ voice: 'tr-TR-AhmetNeural', lang: 'tr-TR' });
const audioBuffer = await tts.synthesize(sesMetni);
fs.writeFileSync(path.join(publicDir, 'ses.mp3'), audioBuffer);

console.log('2/3 -> Remotion Kodları Hazırlanıyor...');

const compositionCode = `import { AbsoluteFill, Audio, Sequence, interpolate, spring, staticFile, useCurrentFrame, useVideoConfig } from 'remotion';
import React from 'react';

const scenes = [
  { year: '1907', title: 'BİR EFSANENİN DOĞUŞU', desc: 'Moda sahilinde yakılan meşale, milyonların sevdası oldu.' },
  { year: '1923', title: 'HARINGTON KUPASI', desc: 'İşgal yıllarında millete umut, cepheye güç veren şanlı zafer.' },
  { year: 'MABED', title: 'ŞÜKRÜ SARACOĞLU RUHU', desc: 'Lefter\\'den Can Bartu\\'ya nesiller boyu süren sarı-lacivert asalet.' },
  { year: 'SONSUZ', title: 'FENERBAHÇE YIKILMAZ', desc: 'Dünden yarına, sarı-lacivert aşkla daima yanında!' },
];

export const MyComp: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  const pulse = interpolate(Math.sin(frame / 12), [-1, 1], [0.85, 1.15]);
  const rot = interpolate(frame, [0, 600], [0, 180]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#001A3A', overflow: 'hidden', fontFamily: 'Segoe UI, Arial, sans-serif' }}>
      <Audio src={staticFile('ses.mp3')} />

      <div style={{
        position: 'absolute',
        width: 1300,
        height: 1300,
        borderRadius: '50%',
        background: 'radial-gradient(circle, rgba(255,237,0,0.22) 0%, rgba(0,26,58,0) 70%)',
        top: '50%',
        left: '50%',
        transform: \`translate(-50%, -50%) scale(\${pulse}) rotate(\${rot}deg)\`,
      }} />

      {scenes.map((s, i) => {
        const start = i * 150;
        const localFrame = frame - start;
        const opacity = interpolate(localFrame, [0, 15, 135, 150], [0, 1, 1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
        const scale = spring({ frame: localFrame, fps, config: { damping: 12 } });

        return (
          <Sequence key={i} from={start} durationInFrames={150}>
            <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center', opacity, transform: \`scale(\${scale})\`, textAlign: 'center', padding: '0 50px' }}>
              <div style={{
                backgroundColor: '#FFED00',
                color: '#001A3A',
                padding: '8px 26px',
                borderRadius: 30,
                fontSize: 26,
                fontWeight: 900,
                letterSpacing: 4,
                marginBottom: 24,
                boxShadow: '0 6px 20px rgba(0,0,0,0.5)',
              }}>
                {s.year}
              </div>
              <h1 style={{
                fontSize: 72,
                color: '#FFFFFF',
                fontWeight: 900,
                margin: '0 0 20px 0',
                letterSpacing: 2,
                textShadow: '0 8px 24px rgba(0,0,0,0.9)',
              }}>
                {s.title}
              </h1>
              <p style={{
                fontSize: 32,
                color: '#E6E6E6',
                maxWidth: 950,
                lineHeight: 1.4,
                margin: 0,
                textShadow: '0 4px 12px rgba(0,0,0,0.9)',
              }}>
                {s.desc}
              </p>
            </AbsoluteFill>
          </Sequence>
        );
      })}
    </AbsoluteFill>
  );
};
`;
fs.writeFileSync(path.join(srcDir, 'Composition.tsx'), compositionCode);

const rootCode = `import { Composition } from 'remotion';
import { MyComp } from './Composition';

export const RemotionRoot: React.FC = () => {
  return (
    <Composition
      id="MyComp"
      component={MyComp}
      durationInFrames={600}
      fps={30}
      width={1920}
      height={1080}
    />
  );
};
`;
fs.writeFileSync(path.join(srcDir, 'Root.tsx'), rootCode);

console.log('3/3 -> Video Render Ediliyor...');
execSync('npx remotion render MyComp out/fenerbahce_hazir.mp4 --concurrency=1', { stdio: 'inherit' });

console.log('\\n>>> Video hazır: out/fenerbahce_hazir.mp4');
execSync('start out/fenerbahce_hazir.mp4');
